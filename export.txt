function addRecord(type){
  const fieldDefinitions={
    users:[["name","Full name"],["email","Email"],["role","Role"]],
    products:[["name","Product name"],["category","Category"],["hs","HS code"],["price","Unit price"]],
    countries:[["name","Country name"],["region","Region"],["currency","Currency"],["tariff","Indicative tariff"]],
    data:[["country","Country"],["product","Product"],["demand","Demand (0–100)"],["growth","Growth (0–100)"],["competition","Competition (0–100)"],["friendly","Import friendliness (0–100)"]]
  };
  const fields=fieldDefinitions[type];
  if(!fields){alert("Unknown record type.");return;}

  const form=`<form id="newform"><div class="form-grid">${fields.map(([key,label])=>`
    <div class="field"><label>${label.toUpperCase()}</label>
    ${type==="data"&&(key==="country"||key==="product")
      ?`<select name="${key}">${(key==="country"?countries.map(c=>c.name):products.map(p=>p.name)).map(x=>`<option>${safe(x)}</option>`).join("")}</select>`
      :`<input name="${key}" ${["price","tariff","demand","growth","competition","friendly"].includes(key)?'type="number" step="any" min="0" max="100"':'type="text"'} required>`}
    </div>`).join("")}</div>
    <br><button class="btn" type="submit">Save Record</button>
    <button class="btn secondary" type="button" onclick="document.getElementById('newform').remove()">Cancel</button>
  </form>`;

  const host=document.getElementById("content");
  const old=document.getElementById("newform");
  if(old)old.closest(".panel")?.remove();

  const box=document.createElement("div");
  box.className="panel";
  box.innerHTML=`<div class="panel-head">Add New Record</div><div class="panel-body">${form}</div>`;
  host.prepend(box);

  document.getElementById("newform").addEventListener("submit",e=>{
    e.preventDefault();
    const d=Object.fromEntries(new FormData(e.target).entries());

    if(type==="users"){
      users.push({...d,status:"Active"});
    }else if(type==="products"){
      d.price=Number(d.price);
      products.push(d);
    }else if(type==="countries"){
      d.tariff=Number(d.tariff);
      countries.push(d);
    }else if(type==="data"){
      ["demand","growth","competition","friendly"].forEach(k=>d[k]=Math.min(100,Number(d[k])));
      market.push(d);
    }
    render();
  });
}

function removeRecord(type,i){
  if(!confirm("Delete this record?"))return;
  if(type==="users")users.splice(i,1);
  if(type==="products")products.splice(i,1);
  if(type==="countries")countries.splice(i,1);
  if(type==="market")market.splice(i,1);
  render();
}

function downloadReport(){
  const type=document.getElementById("reporttype").value;
  const format=document.getElementById("reportformat").value;
  let rows=[],filename=type;

  if(type==="products"){
    rows=[["Product","Category","HS code","Unit price"],...products.map(p=>[p.name,p.category,p.hs,p.price])];
  }else if(type==="countries"){
    rows=[["Country","Region","Currency","Indicative tariff"],...countries.map(c=>[c.name,c.region,c.currency,c.tariff])];
  }else if(type==="market"){
    rows=[["Country","Product","Demand","Growth","Competition","Import friendliness","Score"],...market.map(d=>[d.country,d.product,d.demand,d.growth,d.competition,d.friendly,score(d)])];
  }else if(type==="recommend"){
    rows=[["Rank","Country","Product","Suitability"],...[...market].sort((a,b)=>score(b)-score(a)).map((d,i)=>[i+1,d.country,d.product,score(d)])];
  }else{
    rows=[["ExportInsight Executive Market Summary"],["Users",users.length],["Products",products.length],["Countries",countries.length],["Market records",market.length],[],["Rank","Country","Product","Suitability"],...[...market].sort((a,b)=>score(b)-score(a)).map((d,i)=>[i+1,d.country,d.product,score(d)])];
  }

  const text=rows.map(r=>r.map(v=>format==="csv"?'"'+String(v??"").replace(/"/g,'""')+'"':String(v??"")).join(format==="csv"?",":" | ")).join("\r\n");
  const blob=new Blob(["\ufeff"+text],{type:format==="csv"?"text/csv;charset=utf-8":"text/plain;charset=utf-8"});
  const a=document.createElement("a");
  const url=URL.createObjectURL(blob);
  a.href=url;
  a.download="exportinsight-"+filename+"."+format;
  document.body.appendChild(a);
  a.click();
  a.remove();
  setTimeout(()=>URL.revokeObjectURL(url),1000);
}
